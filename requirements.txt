import os
import time
import logging
import requests
from datetime import datetime, time as dt_time
from bs4 import BeautifulSoup
import schedule
import pytz
from telegram import Bot
from dotenv import load_dotenv

load_dotenv()

# ============ CONFIG ============
TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN")
CHAT_ID = os.getenv("CHAT_ID")
MIN_OI_CHANGE = 5.0          # % OI change
MAX_GAP = 1.0                # % gap allowed
SCAN_INTERVAL = 3            # minutes

logging.basicConfig(level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s")
logger = logging.getLogger(__name__)

bot = Bot(token=TELEGRAM_TOKEN)
IST = pytz.timezone("Asia/Kolkata")

def is_market_hours():
    now = datetime.now(IST).time()
    return dt_time(9, 20) <= now <= dt_time(15, 25)

def safe_request(url, retries=3):
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
    }
    for _ in range(retries):
        try:
            r = requests.get(url, headers=headers, timeout=10)
            if r.status_code == 200:
                return r
        except Exception as e:
            logger.warning(f"Request failed: {e}")
            time.sleep(2)
    return None

def get_fno_data():
    """Fetch OI Spurts data from NiftyTrader"""
    url = "https://www.niftytrader.in/oi-spurts"
    response = safe_request(url)
    if not response:
        return []

    soup = BeautifulSoup(response.text, "lxml")
    stocks = []

    try:
        rows = soup.select("table tbody tr")
        for row in rows[:30]:  # top 30
            cols = row.find_all("td")
            if len(cols) < 8:
                continue

            symbol = cols[0].get_text(strip=True)
            ltp = cols[2].get_text(strip=True)
            price_chg = cols[3].get_text(strip=True)
            oi_chg = cols[6].get_text(strip=True)
            signal = cols[8].get_text(strip=True) if len(cols) > 8 else ""

            # Clean numbers
            try:
                price_chg_val = float(price_chg.replace("%", "").split()[0])
                oi_chg_val = float(oi_chg.replace("%", "").replace(",", ""))
            except:
                continue

            stocks.append({
                "symbol": symbol,
                "ltp": ltp,
                "price_chg": price_chg_val,
                "oi_chg": oi_chg_val,
                "signal": signal
            })
    except Exception as e:
        logger.error(f"Parsing error: {e}")

    return stocks

def send_alert(message):
    try:
        bot.send_message(chat_id=CHAT_ID, text=message, parse_mode="Markdown")
        logger.info("Alert sent")
    except Exception as e:
        logger.error(f"Telegram error: {e}")

def scan_and_alert():
    if not is_market_hours():
        logger.info("Outside market hours")
        return

    logger.info("Scanning started...")
    stocks = get_fno_data()

    if not stocks:
        logger.warning("No data fetched")
        return

    alerts = []
    for s in stocks:
        # Simple filters
        if abs(s["price_chg"]) > MAX_GAP:
            continue
        if s["oi_chg"] < MIN_OI_CHANGE:
            continue

        bias = ""
        if "Long" in s["signal"] and s["price_chg"] > 0:
            bias = "🟢 Buy Call"
        elif "Short" in s["signal"] and s["price_chg"] < 0:
            bias = "🔴 Buy Put"
        else:
            continue

        msg = (
            f"*{s['symbol']}*\n"
            f"Price: {s['ltp']} ({s['price_chg']:+.2f}%)\n"
            f"OI Change: {s['oi_chg']:+.2f}%\n"
            f"Signal: {s['signal']}\n"
            f"Bias: {bias}"
        )
        alerts.append(msg)

    if alerts:
        final_msg = "📊 *F&O High Probability Alerts*\n\n" + "\n\n".join(alerts[:6])
        send_alert(final_msg)
    else:
        logger.info("No stocks matched criteria")

def main():
    logger.info("Bot started successfully")
    send_alert("✅ F&O Alert Bot is LIVE now!")

    schedule.every(SCAN_INTERVAL).minutes.do(scan_and_alert)

    while True:
        schedule.run_pending()
        time.sleep(30)

if __name__ == "__main__":
    main()
